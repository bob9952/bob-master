```python
import os
import random
import csv
import time
from src.parser import parse_instance_file
from src.ga import GeneticAlgorithm
from src.heuristic import first_fit
from src.solution_manager import SolutionManager

# Settings
SAMPLES_PER_FOLDER = 3  # Number of instances to run per subfolder (e.g., 3 from u120, 3 from t60...)
POP_SIZE = 50           # Smaller pop for faster full-sweep
GENERATIONS = 100
RUNS_PER_INSTANCE = 1   # Can increase to 5 or 10 for statistical significance later

def solve_ffd(items_dict, bin_capacity):
    start = time.time()
    sorted_items = sorted(items_dict.keys(), key=lambda k: items_dict[k], reverse=True)
    bins = first_fit(sorted_items, items_dict, bin_capacity)
    duration = time.time() - start
    return len(bins), duration

def main():
    # Use default SolutionManager path which looks for data/Solutions.xlsx
    sol_manager = SolutionManager()
    
    # Robust pathing relative to this script
    script_dir = os.path.dirname(os.path.abspath(__file__))
    
    # Output CSV
    output_file = os.path.join(script_dir, "experiment_results.csv")
    base_data_dir = os.path.join(script_dir, "data")
    
    # Collect all instance files
    all_files = []
    for root, dirs, files in os.walk(base_data_dir):
        txt_files = [f for f in files if f.endswith(".txt") and "binpack" not in f] # Skip the aggregate binpack1-8 files if individual ones exist
        
        # If we are in a leaf folder with txt files
        if txt_files:
            # Randomly sample to avoid running 1000s
            sample = random.sample(txt_files, min(len(txt_files), SAMPLES_PER_FOLDER))
            for f in sample:
                all_files.append(os.path.join(root, f))

    print(f"Found {len(all_files)} instances to test (Sampled {SAMPLES_PER_FOLDER} per folder).")
    
    # Header for Console Output
    print("\n" + "="*115)
    print(f"{'Dataset':<15} | {'Instance':<25} | {'Opt':<5} | {'FFD':<5} | {'GA':<5} | {'Time':<6} | {'Gap GA':<6}")
    print("="*115)

    # Prepare CSV
    with open(output_file, 'w', newline='') as csvfile:
        writer = csv.writer(csvfile)
        writer.writerow(["Dataset", "Instance", "Items", "Capacity", "Optimal", "FFD_Bins", "FFD_Time", "GA_Bins", "GA_Time", "Gap_FFD", "Gap_GA"])
        
        for i, file_path in enumerate(all_files):
            # Parse
            instances = parse_instance_file(file_path)
            if not instances: continue
            instance = instances[0] # Single file = single instance
            
            name = instance['name']
            capacity = instance['capacity']
            items = instance['items_dict']
            num_items = len(items)
            
            # Determine Dataset Name from path (e.g., "Scholl_1")
            folder_name = os.path.basename(os.path.dirname(file_path))
            
            # Get Optimal
            optimal = instance['best_known']
            if not optimal:
                optimal = sol_manager.get_optimal(name)
            
            # 1. FFD
            ffd_res, ffd_time = solve_ffd(items, capacity)
            
            # 2. GA
            start_ga = time.time()
            ga = GeneticAlgorithm(items, capacity, pop_size=POP_SIZE, generations=GENERATIONS, mutation_prob=0.2)
            best_sol = ga.run()
            ga_res = best_sol.num_bins
            ga_time = time.time() - start_ga
            
            # Gaps
            gap_ffd = (ffd_res - optimal) if optimal else "N/A"
            gap_ga = (ga_res - optimal) if optimal else "N/A"
            
            # Print to Console (Formatted Table)
            print(f"{folder_name:<15} | {name:<25} | {optimal:<5} | {ffd_res:<5} | {ga_res:<5} | {ga_time:<6.2f} | {gap_ga:<6}")

            writer.writerow([folder_name, name, num_items, capacity, optimal, ffd_res, ffd_time, ga_res, ga_time, gap_ffd, gap_ga])
            csvfile.flush() # Save progress

    print("\n" + "="*115)
    print(f"\nExperiments completed. Results saved to {output_file}")

if __name__ == "__main__":
    main()
```

```python
def first_fit(item_order, items_dict, bin_capacity):
    """
    Applies the First Fit heuristic to pack items in the given order.
    """
    bins = []
    EPSILON = 1e-6 
    
    for item_id in item_order:
        item_size = items_dict[item_id]
        placed = False
        
        for bin in bins:
            if bin['used'] + item_size <= bin_capacity + EPSILON:
                bin['used'] += item_size
                bin['items'].append((item_id, item_size))
                placed = True
                break
        
        if not placed:
            bins.append({'used': item_size, 'items': [(item_id, item_size)]})
            
    return bins

def next_fit(item_order, items_dict, bin_capacity):
    """
    Applies the Next Fit heuristic.
    Only checks the current (last open) bin.
    """
    bins = []
    EPSILON = 1e-6 
    
    # Start with one empty bin if there are items
    if not item_order:
        return []
        
    # Create first bin
    current_bin = {'used': 0.0, 'items': []}
    bins.append(current_bin)
    
    for item_id in item_order:
        item_size = items_dict[item_id]
        
        # Check only the last bin
        if current_bin['used'] + item_size <= bin_capacity + EPSILON:
            current_bin['used'] += item_size
            current_bin['items'].append((item_id, item_size))
        else:
            # Close current, open new
            current_bin = {'used': item_size, 'items': [(item_id, item_size)]}
            bins.append(current_bin)
            
    return bins

def best_fit(item_order, items_dict, bin_capacity):
    """
    Applies the Best Fit heuristic.
    Places item in the bin where it fits with the MINIMUM remaining space.
    """
    bins = []
    EPSILON = 1e-6 
    
    for item_id in item_order:
        item_size = items_dict[item_id]
        
        best_bin_idx = -1
        min_residual_space = float('inf')
        
        # Check all bins to find tightest fit
        for i, bin in enumerate(bins):
            residual = bin_capacity - bin['used']
            # If it fits...
            if residual >= item_size - EPSILON:
                # ...and is tighter than what we found so far
                if residual < min_residual_space:
                    min_residual_space = residual
                    best_bin_idx = i
        
        if best_bin_idx != -1:
            bins[best_bin_idx]['used'] += item_size
            bins[best_bin_idx]['items'].append((item_id, item_size))
        else:
            bins.append({'used': item_size, 'items': [(item_id, item_size)]})
            
    return bins

def max_rest(item_order, items_dict, bin_capacity):
    """
    Applies the Max-Rest (Worst-Fit) heuristic.
    Places item in the bin with the MAXIMUM remaining space (if it fits).
    """
    bins = []
    EPSILON = 1e-6 
    
    for item_id in item_order:
        item_size = items_dict[item_id]
        
        best_bin_idx = -1
        max_residual_space = -1.0
        
        # Check all bins to find the one with most room
        for i, bin in enumerate(bins):
            residual = bin_capacity - bin['used']
            
            # If it fits...
            if residual >= item_size - EPSILON:
                # ...and has more space than what we found so far
                if residual > max_residual_space:
                    max_residual_space = residual
                    best_bin_idx = i
        
        if best_bin_idx != -1:
            bins[best_bin_idx]['used'] += item_size
            bins[best_bin_idx]['items'].append((item_id, item_size))
        else:
            bins.append({'used': item_size, 'items': [(item_id, item_size)]})
            
    return bins
```
